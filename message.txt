    <script>
      const STORAGE_KEYS = {
        users: 'botforge_users',
        currentUser: 'botforge_current_user',
        bots: 'botforge_bots'
      };

      const state = {
        authMode: 'signin',
        selectedFiles: []
      };

      const getUsers = () => JSON.parse(localStorage.getItem(STORAGE_KEYS.users) || '[]');
      const setUsers = (users) => localStorage.setItem(STORAGE_KEYS.users, JSON.stringify(users));
      const getCurrentUser = () => JSON.parse(localStorage.getItem(STORAGE_KEYS.currentUser) || 'null');
      const setCurrentUser = (user) => localStorage.setItem(STORAGE_KEYS.currentUser, JSON.stringify(user));
      const getBots = () => JSON.parse(localStorage.getItem(STORAGE_KEYS.bots) || '[]');
      const setBots = (bots) => localStorage.setItem(STORAGE_KEYS.bots, JSON.stringify(bots));

      const showToast = (message) => {
        const toast = document.getElementById('toast');
        if (!toast) return;
        toast.textContent = message;
        toast.classList.remove('hidden');
        setTimeout(() => toast.classList.add('hidden'), 2600);
      };

      const resetAuthForm = () => {
        const name = document.getElementById('auth-name');
        const email = document.getElementById('auth-email');
        const pass = document.getElementById('auth-password');
        if (name) name.value = '';
        if (email) email.value = '';
        if (pass) pass.value = '';
      };

      const openAuthModal = (mode = 'signin') => {
        const modal = document.getElementById('auth-modal');
        if (!modal) return;
        state.authMode = mode;
        const title = document.getElementById('auth-title');
        const nameField = document.getElementById('auth-name');

        if (title) title.textContent = mode === 'signup' ? 'Sign up' : 'Sign in';
        if (nameField) nameField.style.display = mode === 'signup' ? 'grid' : 'none';

        document.querySelectorAll('.auth-mode').forEach((button) => {
          button.classList.toggle('active', button.dataset.mode === mode);
        });

        modal.classList.remove('hidden');
      };

      const closeAuthModal = () => {
        const modal = document.getElementById('auth-modal');
        if (modal) modal.classList.add('hidden');
        resetAuthForm();
      };

      const setTheme = (theme) => {
        document.body.setAttribute('data-theme', theme);
        localStorage.setItem('botforge-theme', theme);
        const toggle = document.getElementById('theme-toggle');
        if (toggle) toggle.textContent = theme === 'dark' ? '🌙' : '☀️';
      };

      const providerLogin = (provider) => {
        const displayName = {
          google: 'Google User',
          apple: 'Apple User',
          discord: 'Discord User'
        }[provider] || 'User';

        const email = `${provider}@botforge.com`;
        const users = getUsers();
        let user = users.find((entry) => entry.email === email);

        if (!user) {
          user = {
            id: crypto.randomUUID(),
            name: displayName,
            email,
            password: 'provider-auth',
            provider
          };
          users.push(user);
          setUsers(users);
        }

        setCurrentUser(user);
        renderLoggedInState();
        closeAuthModal();
        showToast(`Signed in with ${provider}`);
      };

      const handleAuthSubmit = (event) => {
        event.preventDefault();
        const name = document.getElementById('auth-name')?.value.trim() || '';
        const email = document.getElementById('auth-email')?.value.trim() || '';
        const password = document.getElementById('auth-password')?.value.trim() || '';

        if (!email || !password) {
          showToast('Please fill in all required fields.');
          return;
        }

        const users = getUsers();

        if (state.authMode === 'signup') {
          if (users.some((user) => user.email.toLowerCase() === email.toLowerCase())) {
            showToast('An account with this email already exists.');
            return;
          }

          const user = {
            id: crypto.randomUUID(),
            name: name || 'New user',
            email,
            password,
            provider: 'local'
          };

          users.push(user);
          setUsers(users);
          setCurrentUser(user);
          renderLoggedInState();
          closeAuthModal();
          showToast('Account created!');
          return;
        }

        const user = users.find((entry) => entry.email.toLowerCase() === email.toLowerCase() && entry.password === password);
        if (!user) {
          showToast('Incorrect email or password.');
          return;
        }

        setCurrentUser(user);
        renderLoggedInState();
        closeAuthModal();
        showToast('Signed in successfully.');
      };

      const renderLoggedInState = () => {
        const user = getCurrentUser();
        const appShell = document.getElementById('app-shell');
        const pageButtons = document.querySelectorAll('.provider-login, #open-signin, #hero-signup');

        pageButtons.forEach((button) => {
          button.disabled = !!user;
          button.style.opacity = user ? '0.7' : '1';
        });

        if (user) {
          if (appShell) appShell.classList.remove('hidden');
          document.documentElement.scrollIntoView({ behavior: 'smooth', block: 'start' });
          const appPanel = document.getElementById('app-shell');
          if (appPanel) appPanel.scrollIntoView({ behavior: 'smooth', block: 'start' });
        } else {
          if (appShell) appShell.classList.add('hidden');
        }

        renderBots();
      };

      const renderBots = () => {
        const list = document.getElementById('bot-list');
        const bots = getBots();
        const user = getCurrentUser();

        if (!list) return;
        if (!user) {
          list.innerHTML = '<p class="empty-state">Log in to deploy your first bot.</p>';
          return;
        }

        const userBots = bots.filter((bot) => bot.ownerId === user.id);

        if (!userBots.length) {
          list.innerHTML = '<div class="empty-state">No bots deployed yet. Upload your project files and deploy your first bot.</div>';
          return;
        }

        list.innerHTML = userBots.map((bot) => `
          <div class="bot-card">
            <div class="bot-head">
              <h4>${bot.name}</h4>
              <span class="bot-status ${bot.running ? 'status-running' : 'status-stopped'}">${bot.running ? 'Running' : 'Stopped'}</span>
            </div>
            <div class="bot-meta">Command: ${bot.command}</div>
            <div class="bot-meta">Files: ${bot.files.length}</div>
            <div class="bot-meta">Deployment ID: ${bot.id.slice(0, 8)}</div>
            <div class="bot-actions">
              <button class="secondary-btn" type="button" data-action="toggle" data-id="${bot.id}">${bot.running ? 'Stop' : 'Start'}</button>
              <button class="secondary-btn" type="button" data-action="logs" data-id="${bot.id}">Logs</button>
              <button class="secondary-btn danger" type="button" data-action="delete" data-id="${bot.id}">Delete</button>
            </div>
          </div>
        `).join('');
      };

      const handleDeploySubmit = (event) => {
        event.preventDefault();
        const user = getCurrentUser();
        if (!user) {
          showToast('Please sign in first.');
          return;
        }

        const name = document.getElementById('bot-name').value.trim();
        const command = document.getElementById('bot-command').value.trim() || 'node index.js';

        if (!name) {
          showToast('Give your bot a name.');
          return;
        }

        const bots = getBots();
        const newBot = {
          id: crypto.randomUUID(),
          ownerId: user.id,
          name,
          command,
          files: state.selectedFiles.map((file) => ({
            name: file.name,
            size: file.size,
            type: file.type || 'unknown'
          })),
          running: true,
          createdAt: new Date().toISOString()
        };

        bots.push(newBot);
        setBots(bots);
        renderBots();
        event.target.reset();
        state.selectedFiles = [];
        const fileList = document.getElementById('file-list');
        if (fileList) fileList.innerHTML = '';
        showToast('Bot deployed successfully.');
      };

      const updateSelectedFiles = (files) => {
        const list = document.getElementById('file-list');
        state.selectedFiles = Array.from(files || []);

        if (!list) return;

        if (!state.selectedFiles.length) {
          list.innerHTML = '<span class="file-chip">No files selected</span>';
          return;
        }

        list.innerHTML = state.selectedFiles.map((file) => `
          <span class="file-chip">${file.name}</span>
        `).join('');
      };

      const handleBotActions = (event) => {
        const actionButton = event.target.closest('[data-action]');
        if (!actionButton) return;

        const { action, id } = actionButton.dataset;
        const bots = getBots();
        const index = bots.findIndex((bot) => bot.id === id);

        if (index === -1) return;

        if (action === 'toggle') {
          bots[index].running = !bots[index].running;
          showToast(`${bots[index].name} ${bots[index].running ? 'started' : 'stopped'}.`);
        }

        if (action === 'logs') {
          const bot = bots[index];
          showToast(`${bot.name}: ${bot.running ? 'Bot is online and ready.' : 'Bot is stopped.'}`);
        }

        if (action === 'delete') {
          bots.splice(index, 1);
          showToast('Bot removed.');
        }

        setBots(bots);
        renderBots();
      };

      const logout = () => {
        localStorage.removeItem(STORAGE_KEYS.currentUser);
        renderLoggedInState();
        showToast('Logged out.');
      };

      const init = () => {
        const savedTheme = localStorage.getItem('botforge-theme') || 'dark';
        setTheme(savedTheme);

        document.querySelectorAll('.auth-mode').forEach((button) => {
          button.addEventListener('click', () => {
            state.authMode = button.dataset.mode;
            openAuthModal(state.authMode);
          });
        });

        const authForm = document.getElementById('auth-form');
        if (authForm) authForm.addEventListener('submit', handleAuthSubmit);

        const closeAuthButton = document.getElementById('close-auth');
        if (closeAuthButton) closeAuthButton.addEventListener('click', closeAuthModal);

        const openSigninButton = document.getElementById('open-signin');
        if (openSigninButton) openSigninButton.addEventListener('click', () => openAuthModal('signup'));

        const heroSignupButton = document.getElementById('hero-signup');
        if (heroSignupButton) heroSignupButton.addEventListener('click', () => openAuthModal('signup'));

        const logoutButton = document.getElementById('logout-btn');
        if (logoutButton) logoutButton.addEventListener('click', logout);

        const deployForm = document.getElementById('deploy-form');
        if (deployForm) deployForm.addEventListener('submit', handleDeploySubmit);

        const botFilesInput = document.getElementById('bot-files');
        if (botFilesInput) botFilesInput.addEventListener('change', (event) => updateSelectedFiles(event.target.files));

        const botList = document.getElementById('bot-list');
        if (botList) botList.addEventListener('click', handleBotActions);

        document.querySelectorAll('.provider-login').forEach((button) => {
          button.addEventListener('click', () => providerLogin(button.dataset.provider));
        });

        const themeToggle = document.getElementById('theme-toggle');
        if (themeToggle) {
          themeToggle.addEventListener('click', () => {
            const nextTheme = document.body.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
            setTheme(nextTheme);
          });
        }

        document.querySelectorAll('.tab-btn').forEach((button) => {
          button.addEventListener('click', () => {
            document.querySelectorAll('.tab-btn').forEach((btn) => btn.classList.remove('active'));
            button.classList.add('active');
          });
        });

        const authModal = document.getElementById('auth-modal');
        if (authModal) {
          authModal.addEventListener('click', (event) => {
            if (event.target.id === 'auth-modal') closeAuthModal();
          });
        }

        renderLoggedInState();
        updateSelectedFiles([]);
      };

      init();
    </script>
  </body>
